二、环境准备

2.1 启动 Minikube 集群

# 启动集群（资源充足，建议 4C8G）
minikube start --cpus=4 --memory=8192 --disk-size=40g --driver=docker --kubernetes-version=v1.28.0

# 查看集群状态
minikube status
kubectl cluster-info

# 启用 ingress addon（关键步骤）
minikube addons enable ingress
minikube addons list | grep ingress

# 启用 metrics-server（用于 HPA 进阶任务）
minikube addons enable metrics-server

2.2 加载本地镜像到 Minikube

# 进入 Minikube 节点 docker 环境（共享宿主机镜像缓存）
eval $(minikube docker-env)

# 构建后端镜像
cd /path/to/ctts/backend
docker build -t backend:ctts-v1 .

# 构建前端镜像
cd /path/to/ctts/frontend
docker build -t frontend:ctts-v1 .

# 拉取 MySQL 镜像
docker pull mysql:8.0

# 验证镜像
docker images | grep -E "backend|frontend|mysql"

# 退出 minikube docker-env
eval $(minikube docker-env -u)

2.3 配置本地 hosts 解析

获取 Minikube IP 后，将其添加到本机 hosts 文件：

# 获取 Minikube IP
minikube ip
# 假设输出：192.168.49.2

# Windows: 编辑 C:\Windows\System32\drivers\etc\hosts（需管理员权限）
# Linux/Mac: 编辑 /etc/hosts
# 添加以下两行：
192.168.49.2 frontend.ctts.local
192.168.49.2 api.ctts.local

四、MySQL 全套资源清单

文件：mysql.yaml 包含：ConfigMap + Secret + PV + PVC + Deployment + Service

4.1 完整 YAML

---
# ============ 1. ConfigMap：MySQL 配置 ============
apiVersion: v1
kind: ConfigMap
metadata:
  name: mysql-config
  namespace: ctts
  labels:
    app: mysql
data:
  my.cnf: |
    [mysqld]
    character-set-server=utf8mb4
    collation-server=utf8mb4_unicode_ci
    max_connections=200
    max_connect_errors=10
    default-authentication-plugin=mysql_native_password
    [client]
    default-character-set=utf8mb4
    [mysql]
    default-character-set=utf8mb4
---
# ============ 2. Secret：MySQL 密码 ============
apiVersion: v1
kind: Secret
metadata:
  name: mysql-secret
  namespace: ctts
  labels:
    app: mysql
type: Opaque
# 注意：value 必须是 base64 编码，不要在 YAML 中明文写密码
# 生成命令：echo -n 'ctts_root_2024' | base64
data:
  MYSQL_ROOT_PASSWORD: Y3R0c19yb290XzIwMjQ=           # ctts_root_2024
  MYSQL_DATABASE: Y3R0cw==                              # ctts
  MYSQL_USER: Y3R0c191c2Vy                             # ctts_user
  MYSQL_PASSWORD: Y3R0c19wd2RfMjAyNA==                  # ctts_pwd_2024
---
# ============ 3. PV：持久化存储卷 ============
apiVersion: v1
kind: PersistentVolume
metadata:
  name: mysql-pv
  labels:
    app: mysql
    type: local
spec:
  capacity:
    storage: 5Gi
  accessModes:
    - ReadWriteOnce
  persistentVolumeReclaimPolicy: Retain
  storageClassName: standard
  hostPath:
    path: /data/ctts/mysql
---
# ============ 4. PVC：存储卷声明 ============
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: mysql-pvc
  namespace: ctts
  labels:
    app: mysql
spec:
  accessModes:
    - ReadWriteOnce
  storageClassName: standard
  resources:
    requests:
      storage: 5Gi
---
# ============ 5. Deployment：MySQL Pod ============
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mysql
  namespace: ctts
  labels:
    app: mysql
    tier: database
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mysql
  strategy:
    type: Recreate
  template:
    metadata:
      labels:
        app: mysql
        tier: database
    spec:
      containers:
        - name: mysql
          image: mysql:8.0
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 3306
              name: mysql
          env:
            - name: MYSQL_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: mysql-secret
                  key: MYSQL_ROOT_PASSWORD
            - name: MYSQL_DATABASE
              valueFrom:
                secretKeyRef:
                  name: mysql-secret
                  key: MYSQL_DATABASE
            - name: MYSQL_USER
              valueFrom:
                secretKeyRef:
                  name: mysql-secret
                  key: MYSQL_USER
            - name: MYSQL_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: mysql-secret
                  key: MYSQL_PASSWORD
          envFrom:
            - configMapRef:
                name: mysql-config
          volumeMounts:
            - name: mysql-data
              mountPath: /var/lib/mysql
            - name: mysql-config-volume
              mountPath: /etc/mysql/conf.d/my.cnf
              subPath: my.cnf
          resources:
            requests:
              cpu: 250m
              memory: 512Mi
            limits:
              cpu: 500m
              memory: 1Gi
          livenessProbe:
            exec:
              command:
                - sh
                - -c
                - "mysqladmin ping -h 127.0.0.1 -u root -p${MYSQL_ROOT_PASSWORD} --silent"
            initialDelaySeconds: 30
            periodSeconds: 10
            timeoutSeconds: 5
          readinessProbe:
            exec:
              command:
                - sh
                - -c
                - "mysql -h 127.0.0.1 -u root -p${MYSQL_ROOT_PASSWORD} -e 'SELECT 1'"
            initialDelaySeconds: 20
            periodSeconds: 5
            timeoutSeconds: 3
      volumes:
        - name: mysql-data
          persistentVolumeClaim:
            claimName: mysql-pvc
        - name: mysql-config-volume
          configMap:
            name: mysql-config
---
# ============ 6. Service：MySQL 内部访问入口 ============
apiVersion: v1
kind: Service
metadata:
  name: mysql-service
  namespace: ctts
  labels:
    app: mysql
spec:
  type: ClusterIP
  selector:
    app: mysql
  ports:
    - port: 3306
      targetPort: 3306
      protocol: TCP
      name: mysql

4.2 关键字段说明

字段 值 说明
strategy.type Recreate MySQL 有状态，不滚动升级，直接重建
volumeMounts.mysql-data /var/lib/mysql 数据目录挂载到 PVC
livenessProbe mysqladmin ping 探活命令，失败重启 Pod
readinessProbe SELECT 1 就绪检查，通过后才接收流量
envFrom.configMapRef mysql-config 整体注入 ConfigMap 数据
imagePullPolicy IfNotPresent 优先用本地镜像

4.3 Base64 编码速查

# 编码
echo -n 'ctts_root_2024' | base64   # 输出：Y3R0c19yb290XzIwMjQ=
echo -n 'ctts' | base64              # 输出：Y3R0cw==
echo -n 'ctts_user' | base64         # 输出：Y3R0c191c2Vy
echo -n 'ctts_pwd_2024' | base64     # 输出：Y3R0c19wd2RfMjAyNA==

# 解码验证

五、Backend 全套资源清单

文件：backend.yaml 包含：ConfigMap + Secret + Deployment + Service

5.1 完整 YAML

---
# ============ 1. ConfigMap：Backend 应用配置 ============
apiVersion: v1
kind: ConfigMap
metadata:
  name: backend-config
  namespace: ctts
  labels:
    app: backend
data:
  # 数据库连接配置（通过 Service DNS 访问 MySQL）
  SPRING_DATASOURCE_URL: jdbc:mysql://mysql-service:3306/ctts?useUnicode=true&characterEncoding=utf8&useSSL=false&serverTimezone=Asia/Shanghai&allowPublicKeyRetrieval=true
  SPRING_DATASOURCE_DRIVER_CLASS_NAME: com.mysql.cj.jdbc.Driver
  SPRING_DATASOURCE_USERNAME: ctts_user
  # 应用配置
  SERVER_PORT: "8080"
  SPRING_PROFILES_ACTIVE: prod
  SPRING_APPLICATION_NAME: ctts-backend
  LOG_LEVEL: INFO
  # CORS 配置
  CORS_ALLOWED_ORIGINS: http://frontend.ctts.local,http://localhost:8080
---
# ============ 2. Secret：Backend 敏感配置 ============
apiVersion: v1
kind: Secret
metadata:
  name: backend-secret
  namespace: ctts
  labels:
    app: backend
type: Opaque
data:
  # 数据库密码（与 mysql-secret 中 MYSQL_PASSWORD 一致）
  SPRING_DATASOURCE_PASSWORD: Y3R0c19wd2RfMjAyNA==   # ctts_pwd_2024
  # JWT 密钥（演示用，生产应使用更强密钥）
  JWT_SECRET: Y3R0cy1qd3Qtc2VjcmV0LWtleS0yMDI0         # ctts-jwt-secret-key-2024
---
# ============ 3. Deployment：Backend Pod ============
apiVersion: apps/v1
kind: Deployment
metadata:
  name: backend
  namespace: ctts
  labels:
    app: backend
    tier: application
spec:
  replicas: 2
  selector:
    matchLabels:
      app: backend
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0
  template:
    metadata:
      labels:
        app: backend
        tier: application
    spec:
      # 初始化容器：等待 MySQL 就绪后再启动后端
      initContainers:
        - name: wait-for-mysql
          image: busybox:1.36
          command:
            - sh
            - -c
            - "until nc -z mysql-service 3306; do echo 'Waiting for MySQL...'; sleep 3; done; echo 'MySQL is up!'"
      containers:
        - name: backend
          image: backend:ctts-v1
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 8080
              name: http
          # 从 ConfigMap 注入非敏感配置
          envFrom:
            - configMapRef:
                name: backend-config
            - secretRef:
                name: backend-secret
          resources:
            requests:
              cpu: 200m
              memory: 384Mi
            limits:
              cpu: 500m
              memory: 768Mi
          # 就绪探针：SpringBoot Actuator
          readinessProbe:
            httpGet:
              path: /actuator/health
              port: 8080
            initialDelaySeconds: 30
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 3
          # 存活探针
          livenessProbe:
            httpGet:
              path: /actuator/health
              port: 8080
            initialDelaySeconds: 60
            periodSeconds: 20
            timeoutSeconds: 5
            failureThreshold: 3
          # 启动探针（SpringBoot 启动较慢，避免被 liveness 误杀）
          startupProbe:
            httpGet:
              path: /actuator/health
              port: 8080
            periodSeconds: 10
            failureThreshold: 12     # 最多等待 120s 启动
---
# ============ 4. Service：Backend 访问入口 ============
apiVersion: v1
kind: Service
metadata:
  name: backend-service
  namespace: ctts
  labels:
    app: backend
spec:
  type: ClusterIP
  selector:
    app: backend
  ports:
    - port: 8080
      targetPort: 8080
      protocol: TCP
      name: http

5.2 关键字段说明

字段 值 说明
initContainers wait-for-mysql 等待 MySQL 3306 端口可用再启动后端，解决服务依赖
envFrom.secretRef backend-secret 整体注入 Secret（密码、JWT）
envFrom.configMapRef backend-config 整体注入 ConfigMap
readinessProbe /actuator/health SpringBoot Actuator 健康检查
startupProbe failureThreshold: 12 最多等 120s 启动，避免 JVM 预热期被杀
strategy RollingUpdate 滚动升级，maxUnavailable=0 保证零停机
replicas 2 双副本高可用

六、Frontend 全套资源清单

文件：frontend.yaml 包含：ConfigMap（Nginx 配置）+ Deployment + Service

6.1 完整 YAML

---
# ============ 1. ConfigMap：Nginx 配置 ============
apiVersion: v1
kind: ConfigMap
metadata:
  name: frontend-config
  namespace: ctts
  labels:
    app: frontend
data:
  default.conf: |
    server {
        listen       80;
        server_name  localhost;

        # 前端静态资源
        location / {
            root   /usr/share/nginx/html;
            index  index.html;
            try_files $uri $uri/ /index.html;   # Vue Router history 模式
        }

        # 反向代理：/api → backend-service
        location /api/ {
            proxy_pass http://backend-service:8080/api/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_connect_timeout 30s;
            proxy_read_timeout 60s;
        }

        # 反向代理：actuator 监控端点
        location /actuator/ {
            proxy_pass http://backend-service:8080/actuator/;
            proxy_set_header Host $host;
        }

        # 静态资源缓存
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
            root /usr/share/nginx/html;
            expires 30d;
            add_header Cache-Control "public, immutable";
        }

        # gzip 压缩
        gzip on;
        gzip_types text/plain text/css application/json application/javascript text/xml application/xml;
        gzip_min_length 1000;
    }
---
# ============ 2. Deployment：Frontend Pod ============
apiVersion: apps/v1
kind: Deployment
metadata:
  name: frontend
  namespace: ctts
  labels:
    app: frontend
    tier: presentation
spec:
  replicas: 2
  selector:
    matchLabels:
      app: frontend
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 1
  template:
    metadata:
      labels:
        app: frontend
        tier: presentation
    spec:
      containers:
        - name: frontend
          image: frontend:ctts-v1
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 80
              name: http
          volumeMounts:
            - name: nginx-config
              mountPath: /etc/nginx/conf.d/default.conf
              subPath: default.conf
          resources:
            requests:
              cpu: 100m
              memory: 128Mi
            limits:
              cpu: 200m
              memory: 256Mi
          readinessProbe:
            httpGet:
              path: /
              port: 80
            initialDelaySeconds: 5
            periodSeconds: 5
          livenessProbe:
            httpGet:
              path: /
              port: 80
            initialDelaySeconds: 10
            periodSeconds: 10
      volumes:
        - name: nginx-config
          configMap:
            name: frontend-config
---
# ============ 3. Service：Frontend 访问入口 ============
apiVersion: v1
kind: Service
metadata:
  name: frontend-service
  namespace: ctts
  labels:
    app: frontend
spec:
  type: ClusterIP
  selector:
    app: frontend
  ports:
    - port: 80
      targetPort: 80

      name: http

7.3 TLS 证书生成命令

# 生成自签名证书
openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout tls.key \
  -out tls.crt \
  -subj "/CN=*.ctts.local/O=CTTS"

# 创建 TLS Secret
kubectl create secret tls ctts-tls-secret \
  --namespace ctts \
  --key tls.key \
  --cert tls.crt

# 验证
kubectl get secret ctts-tls-secret -n ctts

10.2 Pod Pending 排查

现象：kubectl get pods 显示 Pending。

排查命令：

kubectl describe pod <pod-name> -n ctts
# 重点看 Events 部分

常见原因与解决：

原因 Events 特征 解决方案
资源不足 Insufficient cpu/memory 调小 resources.requests 或扩容 Minikube
PVC 未绑定 pod has unbound immediate PersistentVolumeClaims 检查 PV/PVC 是否存在、storageClassName 是否匹配
NodeSelector 不匹配 0/1 nodes are available 检查 nodeSelector / nodeAffinity
调度器问题 FailedScheduling 检查节点状态 kubectl get nodes

示例修复（PVC 未绑定）：

# 检查 PVC
kubectl get pvc -n ctts
# 若 STATUS 为 Pending：
kubectl describe pvc mysql-pvc -n ctts
# 检查 storageClassName 是否有匹配的 PV

# 检查 PV
kubectl get pv
# 若没有 mysql-pv，则先创建：
kubectl apply -f mysql.yaml   # 包含 PV 定义

10.3 CrashLoopBackOff 排查

现象：Pod 反复启动-崩溃-重启。

排查命令：

# 看当前日志
kubectl logs <pod-name> -n ctts

# 看上一次崩溃的日志（关键）
kubectl logs <pod-name> -n ctts --previous

# 看事件
kubectl describe pod <pod-name> -n ctts

常见原因与解决：

原因 日志特征 解决方案
应用启动失败 Caused by: ... SQLException 检查数据库连接配置（URL/账号密码）
配置错误 Failed to bind properties 检查 ConfigMap 中的字段拼写
探针失败 Liveness probe failed: HTTP probe failed 调大 initialDelaySeconds 或 startupProbe failureThreshold
依赖服务未就绪 Connection refused 检查 initContainer / 就绪顺序
资源限制过小 OOMKilled 调大 resources.limits.memory

示例修复（数据库连接失败）：

# 1. 看日志
kubectl logs <backend-pod> -n ctts --previous | tail -30
# 假设看到：Caused by: java.net.ConnectException: Connection refused (mysql-service:3306)

# 2. 检查 MySQL 是否 Running
kubectl get pod -l app=mysql -n ctts
# 若 MySQL 还在 ContainerCreating，等待即可

# 3. 检查 Service 是否存在
kubectl get svc mysql-service -n ctts
# 若不存在，重新 apply mysql.yaml

# 4. 修复后重启 Backend
kubectl rollout restart deployment/backend -n ctts

10.4 ImagePullBackOff 排查

现象：Pod 显示 ImagePullBackOff 或 ErrImagePull。

排查命令：

kubectl describe pod <pod-name> -n ctts
# 看 Events，会显示拉取镜像失败的详细原因

常见原因与解决：

原因 Events 特征 解决方案
镜像不存在 ImagePullBackOff: Back-off pulling image 检查镜像名拼写、tag 是否正确
私有仓库未认证 Failed to pull image: unauthorized 配置 imagePullSecrets
本地镜像未加载到 Minikube not found 执行 eval $(minikube docker-env) 后重新构建
imagePullPolicy 错误 Always 拉不到远程镜像 改为 IfNotPresent

示例修复（本地镜像未加载）：

# 1. 确认 imagePullPolicy 已设为 IfNotPresent
# 2. 进入 Minikube docker 环境构建镜像
eval $(minikube docker-env)
docker build -t backend:ctts-v1 /path/to/backend
docker build -t frontend:ctts-v1 /path/to/frontend

# 3. 验证镜像在 Minikube 内可用
docker images | grep ctts

# 4. 退出并删除失败的 Pod 让其重建
eval $(minikube docker-env -u)
kubectl delete pod -l app=backend -n ctts

10.5 Service 不可达排查

现象：Pod Running，但 curl <service-name> 失败。

排查命令：

# 1. 检查 Service 是否存在
kubectl get svc -n ctts

# 2. 检查 Service Endpoints
kubectl get endpoints mysql-service -n ctts
# 若 ENDPOINTS 为 <none>，说明 selector 没匹配到 Pod

# 3. 对比 selector 与 Pod labels
kubectl get pod -l app=mysql -n ctts --show-labels
kubectl get svc mysql-service -n ctts -o yaml | grep -A5 selector

# 4. 集群内 DNS 测试
kubectl run debug --image=busybox -it --rm -- nslookup mysql-service.ctts.svc.cluster.local

常见原因：selector 标签不匹配、端口不一致、Pod 未就绪（readinessProbe 失败导致 Endpoints 为空）。

  minReplicas: 2
  maxReplicas: 5
  metrics:
    - type: Resource
      resource:
        name: cpu
        target:
          type: Utilization
          averageUtilization: 70
    - type: Resource

        name: memory
        target:

          averageUtilization: 80
